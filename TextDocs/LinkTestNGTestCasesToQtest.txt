## Step 1: Create a Custom Java Annotation

package com.qtest.annotations;

## Step 2: Create the TestNG Listener
This listener implements ITestListener. It checks your test status on completion, looks for the @QTestRun annotation, and fires a native HTTP request to the qTest API. [3, 4, 5] 
(Make sure you have an HTTP client library like java.net.http available in your project, which is built into Java 11+).

package com.qtest.listeners;
import com.qtest.annotations.QTestRun;import org.testng.ITestListener;import org.testng.ITestResult;import java.net.URI;import java.net.http.HttpClient;import java.net.http.HttpRequest;import java.net.http.HttpResponse;import java.time.Instant;
public class QTestListener implements ITestListener {

    // Configuration Settings (Can also load these from a config.properties file)
    private static final String QTEST_URL = "https://qtestnet.com";
    private static final String API_TOKEN = "YOUR_QTEST_API_TOKEN_HERE";
    private static final String PROJECT_ID = "12345"; 

    @Override
    public void onTestSuccess(ITestResult result) {
        sendResultToQTest(result, "PASSED");
    }

    @Override
    public void onTestFailure(ITestResult result) {
        sendResultToQTest(result, "FAILED");
    }

    @Override
    public void onTestSkipped(ITestResult result) {
        sendResultToQTest(result, "SKIPPED");
    }

    private void sendResultToQTest(ITestResult result, String status) {
        // Look for our custom annotation on the executing test method
        QTestRun qTestAnnotation = result.getMethod()
                                         .getConstructorOrMethod()
                                         .getMethod()
                                         .getAnnotation(QTestRun.class);

        if (qTestAnnotation == null) {
            System.out.println("Skipping qTest sync: No @QTestRun annotation found on " + result.getName());
            return;
        }

        long runId = qTestAnnotation.id();
        String timestamp = Instant.now().toString(); // Formats automatically to ISO-8601 UTC

        // Construct the qTest JSON Payload
        String jsonPayload = "{"
                + "\"status\": \"" + status + "\","
                + "\"exe_start_date\": \"" + timestamp + "\","
                + "\"exe_end_date\": \"" + timestamp + "\","
                + "\"name\": \"Automated Execution via TestNG Listener\""
                + "}";

        try {
            HttpClient client = HttpClient.newHttpClient();
            HttpRequest request = HttpRequest.newBuilder()
                    .uri(URI.create(QTEST_URL + "/api/v3/projects/" + PROJECT_ID + "/test-runs/" + runId + "/auto-test-logs"))
                    .header("Authorization", API_TOKEN)
                    .header("Content-Type", "application/json")
                    .POST(HttpRequest.BodyPublishers.ofString(jsonPayload))
                    .build();

            HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());
            
            if (response.statusCode() == 200 || response.statusCode() == 201) {
                System.out.println("Successfully logged result to qTest Run ID: " + runId);
            } else {
                System.err.println("Failed to log to qTest. Status Code: " + response.statusCode() + " | Response: " + response.body());
            }
        } catch (Exception e) {
            System.err.println("Error sending request to qTest REST API: " + e.getMessage());
        }
    }
}

                + "}";

        try {

## Step 2: Implement the Two-Step Listener Logic
Update QTestListener.java to handle the dynamic creation request before submitting results.
(Note: We use a simple regex tool to extract the newly created run ID from qTest's JSON response so we don't need heavy third-party JSON parser libraries).

package com.qtest.listeners;
import com.qtest.annotations.QTestCase;import org.testng.ITestListener;import org.testng.ITestResult;import java.io.PrintWriter;import java.io.StringWriter;import java.net.URI;import java.net.http.HttpClient;import java.net.http.HttpRequest;import java.net.http.HttpResponse;import java.time.Instant;import java.util.regex.Matcher;import java.util.regex.Pattern;
public class QTestListener implements ITestListener {

    private static final String QTEST_URL = "https://qtestnet.com";
    private static final String API_TOKEN = "YOUR_QTEST_API_TOKEN_HERE";
    private static final String PROJECT_ID = "12345"; 
    private final HttpClient client = HttpClient.newHttpClient();

    @Override
    public void onTestSuccess(ITestResult result) {
        processAndSend(result, "PASSED", "Test executed successfully.");
    }

    @Override
    public void onTestFailure(ITestResult result) {
        String note = "Exception: " + result.getThrowable().getMessage() 
                    + "\n\nStack Trace:\n" + getStackTraceAsString(result.getThrowable());
        processAndSend(result, "FAILED", note);
    }

    @Override
    public void onTestSkipped(ITestResult result) {
        processAndSend(result, "SKIPPED", "Test was skipped.");
    }

    private void processAndSend(ITestResult result, String status, String note) {
        QTestCase qTestCase = result.getMethod()
                                    .getConstructorOrMethod()
                                    .getMethod()
                                    .getAnnotation(QTestCase.class);

        if (qTestCase == null) return;

        long caseId = qTestCase.caseId();
        long cycleId = qTestCase.cycleId();
        String testName = result.getName() + " - Run " + Instant.now().getEpochSecond();

        // Step 1: Create the new Test Run dynamically inside the Test Cycle
        String runId = createTestRun(caseId, cycleId, testName);
        
        if (runId != null) {
            // Step 2: Post the execution status and log trace to the new Run ID
            submitExecutionLog(runId, status, note);
        }
    }

    private String createTestRun(long caseId, long cycleId, String runName) {
        String jsonPayload = "{"
                + "\"name\": \"" + runName + "\","
                + "\"test_case\": { \"id\": " + caseId + " },"
                + "\"parentId\": " + cycleId + ","
                + "\"parentType\": \"test-cycle\""
                + "}";

        try {
            HttpRequest request = HttpRequest.newBuilder()
                    .uri(URI.create(QTEST_URL + "/api/v3/projects/" + PROJECT_ID + "/test-runs"))
                    .header("Authorization", API_TOKEN)
                    .header("Content-Type", "application/json")
                    .POST(HttpRequest.BodyPublishers.ofString(jsonPayload))
                    .build();

            HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());
            
            if (response.statusCode() == 200 || response.statusCode() == 201) {
                // Quick regex to extract the "id":XXXX field from the qTest response
                Matcher matcher = Pattern.compile("\"id\"\\s*:\\s*(\\d+)").matcher(response.body());
                if (matcher.find()) {
                    return matcher.group(1);
                }
            } else {
                System.err.println("Failed to create Test Run. Code: " + response.statusCode() + " | Response: " + response.body());
            }
        } catch (Exception e) {
            System.err.println("Error creating Test Run: " + e.getMessage());
        }
        return null;
    }

    private void submitExecutionLog(String runId, String status, String note) {
        String timestamp = Instant.now().toString();
        String jsonPayload = "{"
                + "\"status\": \"" + status + "\","
                + "\"exe_start_date\": \"" + timestamp + "\","
                + "\"exe_end_date\": \"" + timestamp + "\","
                + "\"note\": \"" + escapeJson(note) + "\""
                + "}";

        try {
            HttpRequest request = HttpRequest.newBuilder()
                    .uri(URI.create(QTEST_URL + "/api/v3/projects/" + PROJECT_ID + "/test-runs/" + runId + "/auto-test-logs"))
                    .header("Authorization", API_TOKEN)
                    .header("Content-Type", "application/json")
                    .POST(HttpRequest.BodyPublishers.ofString(jsonPayload))
                    .build();

            client.send(request, HttpResponse.BodyHandlers.ofString());
        } catch (Exception e) {
            System.err.println("Error posting execution log: " + e.getMessage());
        }
    }

    private String getStackTraceAsString(Throwable throwable) {
        if (throwable == null) return "";
        StringWriter sw = new StringWriter();
        throwable.printStackTrace(new PrintWriter(sw));
        return sw.toString();
    }

    private String escapeJson(String input) {
        if (input == null) return "";
        return input.replace("\\", "\\\\").replace("\"", "\\\"").replace("\n", "\\n").replace("\r", "\\r");
    }
}

## Step 1: Update the Listener to Batch Results
We will use qTest's Bulk Automation Log API (POST /api/v3/projects/{projectId}/auto-test-logs?type=automation). package com.qtest.listeners;
import com.qtest.annotations.QTestCase;import org.testng.ITestContext;import org.testng.ITestListener;import org.testng.ITestResult;import java.io.PrintWriter;import java.io.StringWriter;import java.net.URI;import java.net.http.HttpClient;import java.net.http.HttpRequest;import java.net.http.HttpResponse;import java.time.Instant;import java.util.ArrayList;import java.util.List;
public class QTestListener implements ITestListener {

    private static final String QTEST_URL = "https://qtestnet.com";
    private static final String API_TOKEN = "YOUR_QTEST_API_TOKEN_HERE";
    private static final String PROJECT_ID = "12345";
    private static final String TEST_CYCLE_ID = "998877"; // Target cycle for new runs

    // Thread-safe list to collect logs during execution
    private final List<String> testLogsJson = new ArrayList<>();

    @Override
    public void onTestSuccess(ITestResult result) {
        recordResult(result, "PASSED", "Executed successfully.");
    }

    @Override
    public void onTestFailure(ITestResult result) {
        String note = "Exception: " + result.getThrowable().getMessage() 
                    + "\n\nStack Trace:\n" + getStackTraceAsString(result.getThrowable());
        recordResult(result, "FAILED", note);
    }

    @Override
    public void onTestSkipped(ITestResult result) {
        recordResult(result, "SKIPPED", "Test skipped.");
    }

    private void recordResult(ITestResult result, String status, String note) {
        QTestCase qTestCase = result.getMethod()
                                    .getConstructorOrMethod()
                                    .getMethod()
                                    .getAnnotation(QTestCase.class);

        if (qTestCase == null) return;

        String timestamp = Instant.now().toString();
        // Unique identifier matching the automation content format
        String automationContent = result.getTestClass().getName() + "#" + result.getName();

        // Construct individual log JSON snippet matching qTest bulk schema
        String logJson = "{"
                + "\"test_case_version_id\": " + qTestCase.caseId() + "," // Alternative format for direct case targeting
                + "\"status\": \" " + status + "\","
                + "\"execution_start_date\": \"" + timestamp + "\","
                + "\"execution_end_date\": \"" + timestamp + "\","
                + "\"note\": \"" + escapeJson(note) + "\","
                + "\"automation_content\": \"" + automationContent + "\","
                + "\"name\": \"" + result.getName() + "\""
                + "}";

        synchronized (testLogsJson) {
            testLogsJson.add(logJson);
        }
    }

    // This triggers ONCE after all tests in the <test> tag finish
    @Override
    public void onFinish(ITestContext context) {
        if (testLogsJson.isEmpty()) return;

        // Wrap all individual logs into a single master JSON array
        String finalPayload = "{"
                + "\"test_cycle\": " + TEST_CYCLE_ID + ","
                + "\"test_logs\": [" + String.join(",", testLogsJson) + "]"
                + "}";

        try {
            HttpClient client = HttpClient.newHttpClient();
            HttpRequest request = HttpRequest.newBuilder()
                    .uri(URI.create(QTEST_URL + "/api/v3/projects/" + PROJECT_ID + "/auto-test-logs?type=automation"))
                    .header("Authorization", API_TOKEN)
                    .header("Content-Type", "application/json")
                    .POST(HttpRequest.BodyPublishers.ofString(finalPayload))
                    .build();

            HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());
            System.out.println("Bulk upload status: " + response.statusCode());
        } catch (Exception e) {
            System.err.println("Failed to bulk upload logs: " + e.getMessage());
        }
    }

    private String getStackTraceAsString(Throwable throwable) {
        if (throwable == null) return "";
        StringWriter sw = new StringWriter();
        throwable.printStackTrace(new PrintWriter(sw));
        return sw.toString();
    }

    private String escapeJson(String input) {
        if (input == null) return "";
        return input.replace("\\", "\\\\").replace("\"", "\\\"").replace("\n", "\\n").replace("\r", "\\r");
    }
}

                + "\"name\": \"" + result.getName() + "\""
                + "}";

        testLogsJson.add(logJson);
    }

    @Override
    public void onFinish(ITestContext context) {
        if (testLogsJson.isEmpty()) {
            System.out.println("No qTest results captured to upload.");
            return;
        }

        // Dynamically fetch config parameters from environment variables passed by Jenkins/OS
        String qTestUrl = getEnvOrProperty("QTEST_URL");
        String apiToken = getEnvOrProperty("QTEST_API_TOKEN");
        String projectId = getEnvOrProperty("QTEST_PROJECT_ID");
        String testCycleId = getEnvOrProperty("QTEST_CYCLE_ID");

        if (qTestUrl == null || apiToken == null || projectId == null || testCycleId == null) {
            System.err.println("qTest bulk upload aborted: Missing required environment variables (QTEST_URL, QTEST_API_TOKEN, QTEST_PROJECT_ID, QTEST_CYCLE_ID).");
            return;
        }

        // Construct master batched JSON payload
        String finalPayload = "{"
                + "\"test_cycle\": " + testCycleId + ","
                + "\"test_logs\": [" + String.join(",", testLogsJson) + "]"
                + "}";

        System.out.println("Sending " + testLogsJson.size() + " test results to qTest...");

        try {
            HttpClient client = HttpClient.newHttpClient();
            HttpRequest request = HttpRequest.newBuilder()
                    .uri(URI.create(qTestUrl + "/api/v3/projects/" + projectId + "/auto-test-logs?type=automation"))
                    .header("Authorization", apiToken)
                    .header("Content-Type", "application/json")
                    .POST(HttpRequest.BodyPublishers.ofString(finalPayload))
                    .build();

            HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());
            
            if (response.statusCode() == 200 || response.statusCode() == 201) {
                System.out.println("Successfully uploaded batch results to qTest. Status Code: " + response.statusCode());
            } else {
                System.err.println("Failed to upload batch logs to qTest. Code: " + response.statusCode() + " | Response: " + response.body());
            }
        } catch (Exception e) {
            System.err.println("Error establishing connection to qTest REST API during bulk upload: " + e.getMessage());
        }
    }

    private String getEnvOrProperty(String key) {
        String value = System.getenv(key);
        return (value != null) ? value : System.getProperty(key);
    }

    private String getStackTraceAsString(Throwable throwable) {
        if (throwable == null) return "";
        StringWriter sw = new StringWriter();
        throwable.printStackTrace(new PrintWriter(sw));
        return sw.toString();
    }

    private String escapeJson(String input) {
        if (input == null) return "";
        return input.replace("\\", "\\\\")
                    .replace("\"", "\\\"")
                    .replace("\n", "\\n")
                    .replace("\r", "\\r")

Would you like to know how to enhance the listener to handle parallel test execution safely if you use multi-threaded TestNG runs?

yes

To make your framework completely safe for multi-threaded and parallel test execution (e.g., running tests in parallel via classes, methods, or suites), we need to address potential thread-safety issues.
The previous listener used a Collections.synchronizedList, which protects the list itself from crashing when elements are added simultaneously. However, because tests execute concurrently, individual steps, logs, and timestamps can bleed into each other if not properly isolated.
To fix this, we introduce ThreadLocal to isolate the state of each running test execution thread.
Here is the enhanced, completely parallel-safe listener implementation:
## Updated Parallel-Safe Listener (QTestListener.java)

package com.qtest.listeners;
import com.qtest.annotations.QTestCase;import org.testng.ITestContext;import org.testng.ITestListener;import org.testng.ITestResult;import java.io.PrintWriter;import java.io.StringWriter;import java.net.URI;import java.net.http.HttpClient;import java.net.http.HttpRequest;import java.net.http.HttpResponse;import java.time.Instant;import java.util.ArrayList;import java.util.Collections;import java.util.List;
public class QTestListener implements ITestListener {

    // Thread-safe concurrent list to collect final logs from all parallel threads
    private final List<String> testLogsJson = Collections.synchronizedList(new ArrayList<>());

    // ThreadLocal isolates the start time for each individual test thread
    private final ThreadLocal<Instant> testStartTime = new ThreadLocal<>();

    @Override
    public void onTestStart(ITestResult result) {
        // Capture the precise start time for this specific thread
        testStartTime.set(Instant.now());
    }

    @Override
    public void onTestSuccess(ITestResult result) {
        recordResult(result, "PASSED", "Executed successfully.");
    }

    @Override
    public void onTestFailure(ITestResult result) {
        String note = "Exception: " + result.getThrowable().getMessage() 
                    + "\n\nStack Trace:\n" + getStackTraceAsString(result.getThrowable());
        recordResult(result, "FAILED", note);
    }

    @Override
    public void onTestSkipped(ITestResult result) {
        recordResult(result, "SKIPPED", "Test was skipped.");
    }

    private void recordResult(ITestResult result, String status, String note) {
        try {
            QTestCase qTestCase = result.getMethod()
                                        .getConstructorOrMethod()
                                        .getMethod()
                                        .getAnnotation(QTestCase.class);

            if (qTestCase == null) {
                System.out.println("Skipping qTest logging for " + result.getName() + ": No @QTestCase annotation found.");
                return;
            }

            // Retrieve thread-isolated start time, fall back to current time if missing
            Instant start = testStartTime.get();
            String startTimeStr = (start != null) ? start.toString() : Instant.now().toString();
            String endTimeStr = Instant.now().toString();
            
            String automationContent = result.getTestClass().getName() + "#" + result.getName();

            // Construct individual JSON payload snippet
            String logJson = "{"
                    + "\"test_case_version_id\": " + qTestCase.caseId() + ","
                    + "\"status\": \" " + status + "\","
                    + "\"execution_start_date\": \"" + startTimeStr + "\","
                    + "\"execution_end_date\": \"" + endTimeStr + "\","
                    + "\"note\": \"" + escapeJson(note) + "\","
                    + "\"automation_content\": \"" + automationContent + "\","
                    + "\"name\": \"" + result.getName() + "\""
                    + "}";

            testLogsJson.add(logJson);
        } finally {
            // Clean up ThreadLocal to prevent memory leaks in shared thread pools
            testStartTime.remove();
        }
    }

    @Override
    public void onFinish(ITestContext context) {
        if (testLogsJson.isEmpty()) {
            System.out.println("No qTest results captured to upload.");
            return;
        }

        String qTestUrl = getEnvOrProperty("QTEST_URL");
        String apiToken = getEnvOrProperty("QTEST_API_TOKEN");
        String projectId = getEnvOrProperty("QTEST_PROJECT_ID");
        String testCycleId = getEnvOrProperty("QTEST_CYCLE_ID");

        if (qTestUrl == null || apiToken == null || projectId == null || testCycleId == null) {
            System.err.println("qTest bulk upload aborted: Missing required environment variables.");
            return;
        }

        // Combine all thread results safely into one JSON body
        String finalPayload = "{"
                + "\"test_cycle\": " + testCycleId + ","
                + "\"test_logs\": [" + String.join(",", testLogsJson) + "]"
                + "}";

        System.out.println("[qTest] Sending bulk payload of " + testLogsJson.size() + " test results...");

        try {
            HttpClient client = HttpClient.newHttpClient();
            HttpRequest request = HttpRequest.newBuilder()
                    .uri(URI.create(qTestUrl + "/api/v3/projects/" + projectId + "/auto-test-logs?type=automation"))
                    .header("Authorization", apiToken)
                    .header("Content-Type", "application/json")
                    .POST(HttpRequest.BodyPublishers.ofString(finalPayload))
                    .build();

            HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());
            
            if (response.statusCode() == 200 || response.statusCode() == 201) {
                System.out.println("[qTest] Successfully uploaded batch results. Status Code: " + response.statusCode());
            } else {
                System.err.println("[qTest] Failed bulk log upload. Code: " + response.statusCode() + " | Response: " + response.body());
            }
        } catch (Exception e) {
            System.err.println("[qTest] Error establishing connection during bulk upload: " + e.getMessage());
        }
    }

    private String getEnvOrProperty(String key) {
        String value = System.getenv(key);
        return (value != null) ? value : System.getProperty(key);
    }

    private String getStackTraceAsString(Throwable throwable) {
        if (throwable == null) return "";
        StringWriter sw = new StringWriter();
        throwable.printStackTrace(new PrintWriter(sw));
        return sw.toString();
    }

    private String escapeJson(String input) {
        if (input == null) return "";
        return input.replace("\\", "\\\\")
                    .replace("\"", "\\\"")
                    .replace("\n", "\\n")
                    .replace("\r", "\\r")
                    .replace("\t", "\\t");
    }
}

## Why this is completely thread-safe:

* No Shared Timestamps: By tracking testStartTime inside a ThreadLocal<Instant>, Thread A will never overwrite or pull the execution timestamp belonging to Thread B.
* Atomic List Aggregation: The list tracking the final payloads (testLogsJson) uses a strictly synchronized wrapper ensuring that simultaneous list.add() requests from multi-threaded execution never drop reports or trigger ConcurrentModificationExceptions.
* Clean Bulk Dispatch: When all parallel execution threads complete their code blocks, the single master thread runs onFinish exactly once, delivering an organized array straight to qTest.
