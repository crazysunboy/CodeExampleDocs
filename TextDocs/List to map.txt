import java.util.*;
import java.nio.charset.StandardCharsets;

public class Main {
    public static void main(String[] args) {
        List<String> lst = Arrays.asList("apple", "banana", "apple", "cherry", "banana");

        // Map to store: UUID -> List of Index Positions
        Map<UUID, List<Integer>> indexMap = new HashMap<>();

        for (int i = 0; i < lst.size(); i++) {
            String str = lst.get(i);
            
            // Generates a deterministic UUID based on the string's bytes
            UUID uuid = UUID.nameUUIDFromBytes(str.getBytes(StandardCharsets.UTF_8));

            indexMap.computeIfAbsent(uuid, k -> new ArrayList<>())
                    .add(i);
        }

        indexMap.forEach((uuid, indices) -> 
            System.out.println("UUID: " + uuid + " -> Indices: " + indices));
    }
}
