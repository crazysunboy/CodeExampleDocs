pipeline {
    agent any

    // 1. Define the user dropdown interface
    parameters {
        choice(
            name: 'SUITE_XML', 
            choices: ['testng-regression.xml', 'testng-smoke.xml', 'testng-sanity.xml'], 
            description: 'Select the TestNG XML suite file to run'
        )
    }

    tools {
        maven 'Maven_3.9.x' // Must match Manage Jenkins -> Tools exactly
        jdk 'Java_17'       // Must match Manage Jenkins -> Tools exactly
    }

    stages {
        stage('Clean Environment') {
            steps {
                echo 'Cleaning workspace and terminating any rogue browser instances...'
                cleanWs()
                
                script {
                    if (isUnix()) {
                        sh '''
                            pkill -f chrome || true
                            pkill -f chromedriver || true
                        '''
                    } else {
                        bat '''
                            taskkill /F /IM chrome.exe /T /FI "STATUS eq RUNNING" 2>nul || exit 0
                            taskkill /F /IM chromedriver.exe /T /FI "STATUS eq RUNNING" 2>nul || exit 0
                        '''
                    }
                }
            }
        }

        stage('Checkout Code') {
            steps {
                echo 'Pulling framework source files from SCM...'
                checkout scm
            }
        }

        stage('Execute TestNG Suite') {
            steps {
                // 2. Injecting the parameter dynamically into Maven runtime
                echo "Launching Maven TestNG execution for suite: ${params.SUITE_XML}"
                script {
                    if (isUnix()) {
                        sh "mvn clean test -DsuiteXmlFile=${params.SUITE_XML}"
                    } else {
                        bat "mvn clean test -DsuiteXmlFile=${params.SUITE_XML}"
                    }
                }
            }
        }
    }

    post {
        always {
            echo 'Processing test artifacts and dashboard results...'
            
            junit allowEmptyResults: true, testResults: '**/target/surefire-reports/testng-results.xml'
            
            publishHTML([
                allowMissing: true,
                alwaysLinkToLastBuild: true,
                keepAll: true,
                reportDir: 'target/surefire-reports',
                reportFiles: 'index.html',
                reportName: 'Standard TestNG Report'
            ])
            
            publishHTML([
                allowMissing: true,
                alwaysLinkToLastBuild: true,
                keepAll: true,
                reportDir: 'target/extent-reports', 
                reportFiles: 'Spark.html', 
                reportName: 'Extent Report'
            ])
        }
    }
}
