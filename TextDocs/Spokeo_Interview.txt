import java.io.*;
import java.util.*;

/*

Programming problem:

Given array of integers and an integer number S (target), find all pairs that sum up to the given number S (target).

1. Number can be used only once to be a pair
2. Duplicated pairs are possible
3. Sorting the array is not allowed.

* Example 1
array = [1, 5, 3, 2, 2, 3], target = 4
print [1, 3], [2, 2]

* Example 2
array = [1, 5, 3, 2, 2, 3, 3], target = 4
print [1, 3], [2, 2]

* Example 3
array = [2, 2, 2, 2, 2], target = 4
print [2, 2], [2, 2]


Clarifying questions:

Design discussion:
1. traverse the array
 1.1 Store the idx and value in a map -- might not be required
2. store the index of the first number in a temp set
3. evaluate expr target - arr[idx] - find partner
4. Lookup the partner in array - get the idx
  4.1 if partner found store the idx in the innerset and outerset (lookup) 
  4.2 return the result set with values from inner set idx positions
  4.3 continue the loop if partner does not exist in the array
5. continue with the loop

 */

class Solution {

  public static void main(String[] args) {
    Integer[] arr={2, 2, 2, 2, 2};
    Integer target=4;
    Integer num1=null;
    Integer num2=null;





    Set<Integer> outerSet=new HashSet<>();
    Set<Integer> innerSet=new HashSet<>();
    List<Integer> resultList=new ArrayList<>();

      for(Integer i=0; i<arr.length; i++)  {
        resultList.clear();
        innerSet.clear();        
        innerSet.add(i);

        for (Integer j=i+1; j<arr.length; j++) {
              num1=arr[i]; //1
              num2=target-num1;  //3
              //System.out.println(outerSet);
              if(arr[j]==num2 && innerSet.add(j) && (!outerSet.contains(i)) && (!outerSet.contains(j))){
                if (innerSet.size()==2) {
                  for(Integer idx:innerSet) {
                    resultList.add(arr[idx]);
                  }
                  System.out.println(resultList);
                  
                  outerSet.add(i);
                  outerSet.add(j);
                  //System.out.print("DEBUG");
                  //System.out.println(outerSet);
                  break;
                } 
                
              }
          }
        }
  }
}
